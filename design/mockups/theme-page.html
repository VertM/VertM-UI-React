import { VertMConfigProvider, cobaltTheme } from '@vertm/react';

<VertMConfigProvider
  theme={{
    ...cobaltTheme,
    token: { colorPrimary: '#b23b2e', borderRadius: 4 },
    vertical: { columnSize: 36, columnGap: 16 },
  }}
>
  <App />
</VertMConfigProvider>
